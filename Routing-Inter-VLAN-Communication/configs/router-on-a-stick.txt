! Router-on-a-Stick configuration
! Assumption: Router GigabitEthernet0/0 connects to Core Switch GigabitEthernet0/1.
! Adjust interface names to match your Packet Tracer router.

enable
configure terminal
hostname CORPORATE-R1
no ip domain-lookup

interface gigabitEthernet0/0
 description TRUNK_TO_CORE_SWITCH
 no ip address
